"""One-off SQLite migration: adds the new habit columns to productivity_logs.

Run from the project root with Flask stopped:
    python add_columns.py
Safe to run more than once (existing columns are skipped).
"""
import sqlite3
from pathlib import Path

DB = Path(__file__).resolve().parent / "instance" / "ppia.db"

NEW_COLUMNS = {
    "sleep_hours": "FLOAT",
    "disruptions": "INTEGER",
    "junk_food": "INTEGER",
    "task_difficulty": "INTEGER",
    "free_hours": "FLOAT",
    "day_type": "VARCHAR(16)",
    "weather": "VARCHAR(16)",
}


def main():
    if not DB.exists():
        raise SystemExit(f"Database not found: {DB}")
    con = sqlite3.connect(DB)
    try:
        table = None
        for (name,) in con.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"):
            cols = [r[1] for r in con.execute(f'PRAGMA table_info("{name}")')]
            if "study_hours" in cols:
                table, existing = name, set(cols)
                break
        if table is None:
            raise SystemExit("No table with a study_hours column found.")
        print(f"Table: {table}")
        for col, typ in NEW_COLUMNS.items():
            if col in existing:
                print(f"  skip  {col} (already there)")
            else:
                con.execute(f'ALTER TABLE "{table}" ADD COLUMN {col} {typ}')
                print(f"  added {col} {typ}")
        con.commit()
    finally:
        con.close()
    print("Done.")


if __name__ == "__main__":
    main()

def create_app(config_class=Config):
    app = Flask(__name__)
    app.config.from_object(config_class)

    db.init_app(app)
    login_manager.init_app(app)
    CSRFProtect(app)

    from app import models  # noqa: F401

    from app.main.routes import main_bp
    app.register_blueprint(main_bp)

    from app.auth.routes import auth_bp
    app.register_blueprint(auth_bp)

    from app.insights.routes import insights

        db.create_all()

    return app

from flask_wtf import FlaskForm
from wtforms import StringField, PasswordField, SubmitField
from wtforms.validators import (
    DataRequired, Email, EqualTo, Length, ValidationError
)

from app.models import User


class RegisterForm(FlaskForm):
    username = StringField(
        "Username", validators=[DataRequired(), Length(min=3, max=50)]
    )
    email = StringField(
        "Email", validators=[DataRequired(), Email(), Length(max=120)]
    )
    password = PasswordField(
        "Password", validators=[DataRequired(), Length(min=8, max=128)]
    )
    confirm_password = PasswordField(
        "Confirm password",
        validators=[DataRequired(), EqualTo("password", message="Passwords must match.")],
    )
    submit = SubmitField("Create account")

    def validate_username(self, field):
        if User.query.filter_by(username=field.data).first():
            raise ValidationError("That username is already taken.")

    def validate_email(self, field):
        if User.query.filter_by(email=field.data.lower()).first():
            raise ValidationError("That email is already registered.")


class LoginForm(FlaskForm):
    email = StringField("Email", validators=[DataRequired(), Email()])
    password = PasswordField("Password", validators=[DataRequired()])
    submit = SubmitField("Log in")

from flask import Blueprint, render_template, redirect, url_for, flash, request
from flask_login import login_user, logout_user, login_required, current_user

from app.extensions import db
from app.models import User
from app.auth.forms import RegisterForm, LoginForm

auth_bp = Blueprint("auth", __name__)


@auth_bp.route("/register", methods=["GET", "POST"])
def register():
    if current_user.is_authenticated:
        return redirect(url_for("main.index"))

    form = RegisterForm()
    if form.validate_on_submit():
        user = User(
            username=form.username.data.strip(),
            email=form.email.data.strip().lower(),
        )
        user.set_password(form.password.data)
        db.session.add(user)
        db.session.commit()
        flash("Account created. Please log in.", "success")
        return redirect(url_for("auth.login"))

    return render_template("register.html", form=form)


@auth_bp.route("/login", methods=["GET", "POST"])
def login():
    if current_user.is_authenticated:
        return redirect(url_for("main.index"))

    form = LoginForm()
    if form.validate_on_submit():
        user = User.query.filter_by(email=form.email.data.strip().lower()).first()
        if user and user.check_password(form.password.data):
            login_user(user)
            next_page = request.args.get("next")
            # only follow "next" if it is a path on this site
            if next_page and next_page.startswith("/") and not next_page.startswith("//"):
                return redirect(next_page)
            return redirect(url_for("main.index"))
        flash("Invalid email or password.", "danger")

    return render_template("login.html", form=form)


@auth_bp.route("/logout", methods=["POST"])
@login_required
def logout():
    logout_user()
    flash("You have been logged out.", "info")
    return redirect(url_for("main.index"))

Three kinds of things live here:

  HABITS     goals shown in the weekly grid, heatmap and today panel. Each has a
             "met" rule, so a day is done / not met / no data.
  CONTEXT    daily inputs with no goal (free hours, day type, ...). They are
             logged like habits but never get a grid cell or a "met" state.
  computed   log_streak() and deep_work_days_prev7() are worked out from
             existing entries, so nothing is logged for them.

An input is one of three types:
  number   free numeric input (min / max / step / cast)
  check    one-tap toggle, True = done

# ------------------------------ goal logic ----------------------------------
def is_met(habit, log):
    """True = goal met, False = logged but not met, None = no data."""
    value = getattr(log, habit["field"], None)
    return None if value is None else bool(habit["met"](value))


def goals_met(log):
    return sum(1 for h in HABITS if is_met(h, log))


def has_data(log):
    """True if at least one goal habit was logged on this day."""
    return log is not None and any(is_met(h, log) is not None for h in HABITS)


def level(log):
    """0 = no data, 1-4 = share of goals met (4 = full day)."""
    states = [is_met(h, log) for h in HABITS]
    if all(s is None for s in states):
        return 0
    frac = sum(1 for s in states if s) / len(HABITS)
    return 4 if frac == 1 else 3 if frac >= 0.7 else 2 if frac >= 0.4 else 1


# --------------------------- computed ML features ---------------------------
# logs_by_date: dict of date -> log row for one user.
def log_streak(logs_by_date, today, cap=STREAK_CAP):
    """Consecutive logged days ending YESTERDAY (today is excluded), capped."""
    n, day = 0, today - timedelta(days=1)
    while n < cap and has_data(logs_by_date.get(day)):
        n += 1
        day -= timedelta(days=1)
    return n


def deep_work_days_prev7(logs_by_date, today):
    """How many of the previous 7 days (today excluded) met the deep work goal."""
    study = BY_KEY["study"]
    return sum(
        1 for i in range(1, 8)
        if logs_by_date.get(today - timedelta(days=i)) is not None
        and is_met(study, logs_by_date[today - timedelta(days=i)]) is True
    )

    /insights/log/<id>    "Details": the estimate for one of YOUR logged days

Both use the same 13 model inputs as the home-page card (ml/settings.py
FEATURES). Nothing here trains or changes a model; it only calls predict_day.
"""
from datetime import timedelta

from flask import abort, flash, render_template, request
from flask_login import current_user, login_required

# Same Blueprint object that app/insights/__init__.py creates and registers.
# (Aliased to `insights`, so both `from app.insights import bp` and
# `from app.insights.routes import insights` keep working.)
from app.insights import bp as insights
from app.insights.service import HISTORY_DAYS, LABELS, _inputs
from app.models import ProductivityLog
from ml.predict import (ModelsNotTrainedError, data_stage, model_info,
                        predict_day)
from ml.settings import FEATURES

# label, min, max, step, default for the "try your own numbers" form.
# Only used for the form's hints; predict_day does the real validation.
NUMBER_SPEC = {
    "deep_work": ("Deep work (hours)", 0, 24, 0.5, 3),
    "sleep_hours": ("Sleep (hours)", 0, 24, 0.5, 7),
    "caffeine_level": ("Caffeine level (0-10)", 0, 10, 1, 2),
    "stress_level": ("Stress (1-10)", 1, 10, 1, 5),
    "log_streak": ("Days logged in a row (0-14)", 0, 14, 1, 3),
    "disruptions": ("Disruptions", 0, 20, 1, 1),
    "task_difficulty": ("Task difficulty (1-5)", 1, 5, 1, 3),
    "deep_work_days_prev7": ("Deep-work goal days, last 7 (0-7)", 0, 7, 1, 3),
    "free_hours": ("Free hours", 0, 24, 0.5, 5),
}
CHECK_SPEC = {"journaled": "Journaled today", "junk_food": "Had junk food"}
SELECT_SPEC = {"day_type": "Day type", "weather": "Weather"}

# Friendly names for the details page (service.LABELS lacks the last three)
NAMES = {**LABELS, "journaled": "Journaling", "log_streak": "Days logged in a row",
         "deep_work_days_prev7": "Deep-work goal days (last 7)"}


def _pretty(name):
    return NAMES.get(name) or name.replace("_", " ").capitalize()


def _categories():
    """(categories dict, models_trained). Categories come from training."""
    try:
        return model_info().get("categories", {}), True
    except ModelsNotTrainedError:
        return {}, False


def _fields(values, categories):
    """Form fields in FEATURES order, pre-filled from `values`."""
    fields = []
    for name in FEATURES:
        if name in SELECT_SPEC:
            options = categories.get(name, [])
            fields.append({"name": name, "kind": "select",
                           "label": SELECT_SPEC[name], "options": options,
                           "value": values.get(name, options[0] if options else "")})
        elif name in CHECK_SPEC:
            fields.append({"name": name, "kind": "check",
                           "label": CHECK_SPEC[name],
                           "value": bool(values.get(name, 0))})
        else:
            label, lo, hi, step, default = NUMBER_SPEC.get(
                name, (_pretty(name), None, None, "any", 0))
            fields.append({"name": name, "kind": "number", "label": label,
                           "min": lo, "max": hi, "step": step,
                           "value": values.get(name, default)})
    return fields


def _read_form():
    """Raw values from the submitted form (predict_day validates them)."""
    values = {}
    for name in FEATURES:
        if name in CHECK_SPEC:
            values[name] = 1 if request.form.get(name) else 0
        else:
            values[name] = (request.form.get(name) or "").strip()
    return values


def _check_numbers(values):
    """First problem in the typed numbers as (field, message), else None.

    Friendly wording for the form. predict_day re-checks everything too.
    """
    for name in FEATURES:
        if name not in NUMBER_SPEC:
            continue
        label, lo, hi, _step, _default = NUMBER_SPEC[name]
        short = label.split(" (")[0]
        try:
            number = float(values[name])
        except (TypeError, ValueError):
            return name, f"{short}: enter a number."
        if number != number or not lo <= number <= hi:      # NaN fails too
            return name, f"{short} must be between {lo:g} and {hi:g}."
    return None


@insights.route("/", methods=["GET", "POST"])
@login_required
def index():
    categories, trained = _categories()
    values, result, error, error_field = {}, None, None, None

    if request.method == "POST":
        values = _read_form()
        problem = _check_numbers(values)
        if problem:
            error_field, error = problem
        else:
            try:
                result = predict_day(values)
            except ModelsNotTrainedError:
                trained = False
            except ValueError as exc:
                error = str(exc)

    n_logs = ProductivityLog.query.filter_by(user_id=current_user.id).count()
    return render_template(
        "insights/index.html", fields=_fields(values, categories),
        trained=trained, result=result, error=error, error_field=error_field,
        n_logs=n_logs, stage=data_stage(n_logs))


@insights.route("/log/<int:log_id>")
@login_required
def estimate_log(log_id):
    from app.habits import deep_work_days_prev7, log_streak

    # Only the owner may open a log (404 for anyone else's id)
    log = ProductivityLog.query.filter_by(
        id=log_id, user_id=current_user.id).first_or_404()

    since = log.log_date - timedelta(days=HISTORY_DAYS)
    history = ProductivityLog.query.filter(
        ProductivityLog.user_id == current_user.id,
        ProductivityLog.log_date >= since,
        ProductivityLog.log_date <= log.log_date).all()
    by_date = {r.log_date: r for r in history}

    values, missing = _inputs(log, by_date, log_streak, deep_work_days_prev7)
    ctx = {"log": log, "result": None, "message": None,
           "rows": [], "drivers": []}

    if missing:
        ctx["message"] = ("This day can't be estimated yet. Still missing: "
                          + ", ".join(_pretty(f) for f in missing) + ".")
        return render_template("insights/estimate.html", **ctx)

    try:
        ctx["result"] = predict_day(values)
        info = model_info()
    except ModelsNotTrainedError:
        ctx["message"] = "Models not trained yet. Run: python -m ml.train"
        return render_template("insights/estimate.html", **ctx)
    except ValueError as exc:
        ctx["message"] = str(exc)
        return render_template("insights/estimate.html", **ctx)

    ctx["rows"] = [(_pretty(f), values[f]) for f in FEATURES]
    importance = (info.get("feature_importance") or {}).get("regressor") or {}
    ctx["drivers"] = [(_pretty(k), round(v * 100))
                      for k, v in list(importance.items())[:5]]
    return render_template("insights/estimate.html", **ctx)

    # Only the owner may open a log (404 for anyone else's id)
    log = ProductivityLog.query.filter_by(
        id=log_id, user_id=current_user.id).first_or_404()

    since = log.log_date - timedelta(days=HISTORY_DAYS)
    history = ProductivityLog.query.filter(
        ProductivityLog.user_id == current_user.id,
        ProductivityLog.log_date >= since,
        ProductivityLog.log_date <= log.log_date).all()
    by_date = {r.log_date: r for r in history}

    values, missing = _inputs(log, by_date, log_streak, deep_work_days_prev7)
    ctx = {"log": log, "result": None, "message": None,
           "rows": [], "drivers": []}

    if missing:
        ctx["message"] = ("This day can't be estimated yet. Still missing: "
                          + ", ".join(_pretty(f) for f in missing) + ".")
        return render_template("insights/estimate.html", **ctx)

    try:
        ctx["result"] = predict_day(values)
        info = model_info()
    except ModelsNotTrainedError:
        ctx["message"] = "Models not trained yet. Run: python -m ml.train"
        return render_template("insights/estimate.html", **ctx)
    except ValueError as exc:
        ctx["message"] = str(exc)
        return render_template("insights/estimate.html", **ctx)

    ctx["rows"] = [(_pretty(f), values[f]) for f in FEATURES]
    importance = (info.get("feature_importance") or {}).get("regressor") or {}
    ctx["drivers"] = [(_pretty(k), round(v * 100))
                      for k, v in list(importance.items())[:5]]
    return render_template("insights/estimate.html", **ctx)

"""Small helper so other pages (like the home page) can show an ML estimate.

    from app.insights.service import latest_estimate
    data["estimate"] = latest_estimate(current_user.id)

Returns one of:
    None                                  models not trained, or something broke
                                          (the template then hides the card)
    {"status": "no_log"}                  no log in the last month
    {"status": "incomplete", "message"}   a log exists but inputs are missing;
                                          the message names them
    {"status": "ok", "score", "level", "error", "log_id", "log_date",
                 plus what the home card shows: "range_lo"/"range_hi" (score +-
                 typical error), "probs" (chance of each level), "low_below"/
                 "high_from" (level cut-offs), "levers" (biggest what-ifs),
                 "source" ("today's log", "yesterday's log", ...)}

How the inputs are built
    The model's inputs are named like the DATASET (ml/settings.py FEATURES),
    but the app stores them under different names, so DB_FIELDS maps one to the
    other (deep_work lives in study_hours, caffeine_level in caffeine_mg, ...).
    Two inputs are not stored at all; they are computed from the days before
    the one being estimated (app.habits):
        log_streak            consecutive logged days ending the day before
        deep_work_days_prev7  how many of the previous 7 days met the goal

How the log is chosen
    Logs are created one habit at a time, so today's row is often half empty.
    This looks at the newest 14 logs (last ~month) and uses the newest one that
    has every input. If none is complete, the message names what the NEWEST
    log is missing, so the user knows what to log next. A check habit
    (journaled) that is None means "not done", so it counts as 0. A dropdown
    habit that is None (junk food, day type, weather, ...) is really missing.

Design rule: an EXPERIMENTAL feature must never be able to break the home page,
so every failure is caught here and logged instead of raised.
"""
from datetime import date, timedelta

from flask import current_app

from ml.predict import ModelsNotTrainedError, predict_day, suggest_levers
from ml.settings import FEATURES, LEVELS, LEVEL_BINS

# dataset feature name -> ProductivityLog column
DB_FIELDS = {
    "deep_work": "study_hours",
    "sleep_hours": "sleep_hours",
    "caffeine_level": "caffeine_mg",       # stores the 0-10 LEVEL, not mg
    "stress_level": "stress_level",
    "journaled": "journaled",
    "disruptions": "disruptions",
    "junk_food": "junk_food",
    "task_difficulty": "task_difficulty",
    "day_type": "day_type",
    "weather": "weather",
    "free_hours": "free_hours",
}
COMPUTED = ("log_streak", "deep_work_days_prev7")
CHECK_FEATURES = ("journaled",)            # None means "not done"

LABELS = {
    "deep_work": "Deep work", "sleep_hours": "Sleep",
    "caffeine_level": "Caffeine", "stress_level": "Stress",
    "disruptions": "Disruptions", "junk_food": "Junk food",
    "task_difficulty": "Task difficulty", "day_type": "Day type",
    "weather": "Weather", "free_hours": "Free hours",
}

RECENT_LOGS = 14
HISTORY_DAYS = 32      # enough to compute a 14-day streak for a 14-day-old log


def _inputs(log, by_date, log_streak, deep_work_days_prev7):
    """Return (values, missing) for one log row."""
    values, missing = {}, []
    for f in FEATURES:
        if f == "log_streak":
            values[f] = log_streak(by_date, log.log_date)
        elif f == "deep_work_days_prev7":
            values[f] = deep_work_days_prev7(by_date, log.log_date)
        else:
            v = getattr(log, DB_FIELDS[f], None)
            if f in CHECK_FEATURES:
                v = 1 if v else 0
            if v is None:
                missing.append(f)
                continue
            values[f] = int(v) if isinstance(v, bool) else v
    return values, missing


def _missing_message(log, missing):
    names = ", ".join(LABELS.get(f, f) for f in missing)
    if log.log_date == date.today():
        return f"To estimate today, still log: {names}."
    return (f"Your latest log ({log.log_date.isoformat()}) is missing: "
            f"{names}. Log them to get an estimate.")


def _source(d):
    today = date.today()
    if d == today:
        return "today's log"
    if d == today - timedelta(days=1):
        return "yesterday's log"
    return f"your log from {d.isoformat()}"


def _ok(log, values, result):
    """Everything the home-page card shows for a complete day."""
    score = result["predicted_score"]
    err = result["typical_error_points"]
    d = log.log_date
    try:
        levers = suggest_levers(values)
    except Exception:                       # the extras must never break the card
        current_app.logger.exception("ML levers failed")
        levers = []
    probs = result.get("level_probabilities", {})
    return {
        "status": "ok",
        "score": score,
        "level": result["predicted_level"],
        "error": round(err, 1) if err else None,
        "range_lo": round(max(0.0, score - err), 1) if err else None,
        "range_hi": round(min(100.0, score + err), 1) if err else None,
        "probs": [{"level": lvl, "pct": round(probs.get(lvl, 0) * 100)}
                  for lvl in LEVELS],
        "low_below": LEVEL_BINS[1],
        "high_from": LEVEL_BINS[2],
        "levers": levers,
        "source": _source(d),
        "log_id": log.id,
        "log_date": d.isoformat() if hasattr(d, "isoformat") else str(d),
    }


def latest_estimate(user_id: int):
    try:
        from app.habits import deep_work_days_prev7, log_streak
        from app.models import ProductivityLog   # here to avoid import cycles

        unknown = [f for f in FEATURES
                   if f not in DB_FIELDS and f not in COMPUTED]
        if unknown:
            return {"status": "incomplete",
                    "message": "The app does not know how to build: "
                               + ", ".join(unknown) + " (see DB_FIELDS in "
                               "app/insights/service.py)."}
        absent = sorted({DB_FIELDS[f] for f in FEATURES if f in DB_FIELDS
                         and not hasattr(ProductivityLog, DB_FIELDS[f])})
        if absent:
            return {"status": "incomplete",
                    "message": "The log table has no column named: "
                               + ", ".join(absent) + "."}

        since = date.today() - timedelta(days=HISTORY_DAYS)
        rows = (ProductivityLog.query
                .filter(ProductivityLog.user_id == user_id,
                        ProductivityLog.log_date >= since)
                .order_by(ProductivityLog.log_date.desc())
                .all())
        if not rows:
            return {"status": "no_log"}
        by_date = {r.log_date: r for r in rows}

        newest_missing = None
        for log in rows[:RECENT_LOGS]:
            values, missing = _inputs(log, by_date, log_streak,
                                      deep_work_days_prev7)
            if missing:
                if newest_missing is None:
                    newest_missing = (log, missing)
                continue
            result = predict_day(values)
            return _ok(log, values, result)

        log, missing = newest_missing
        return {"status": "incomplete", "message": _missing_message(log, missing)}
    except ModelsNotTrainedError:
        return None
    except ValueError as exc:                    # value outside the model's range
        return {"status": "incomplete", "message": str(exc)}
    except Exception:                            # never break the home page
        current_app.logger.exception("ML estimate failed")
        return None

def build_dashboard(user):
    today = date.today()

    # Heatmap starts on a Sunday, about one year back
    start = today - timedelta(days=363)
    start -= timedelta(days=(start.weekday() + 1) % 7)

    logs = ProductivityLog.query.filter(
        ProductivityLog.user_id == user.id,
        ProductivityLog.log_date >= start,
    ).all()
    by_date = {l.log_date: l for l in logs}

    weeks, d = [], start
    while d <= today:
        week = []
        for _ in range(7):
            if d > today:
                week.append(None)
            else:
                log = by_date.get(d)
                lvl = level(log) if log else 0
                tip = (f"{d.isoformat()}: {goals_met(log)}/{len(HABITS)} goals"
                       if lvl else f"{d.isoformat()}: no data")
                week.append({
                    "date": d.isoformat(), "level": lvl, "logged": lvl > 0,
                    "today": d == today, "tip": tip,
                })
            d += timedelta(days=1)
        weeks.append(week)

    labels, prev = [], None
    for w in weeks:
        month = date.fromisoformat(w[0]["date"]).strftime("%b")
        labels.append(month if month != prev else "")
        prev = month

    # This week, Monday to Sunday
    monday = today - timedelta(days=today.weekday())
    week_dates = [monday + timedelta(days=i) for i in range(7)]
    today_log = by_date.get(today)

    habits = []
    for h in HABITS:
        days = []
        for wd in week_dates:
            if wd > today or wd not in by_date:
                days.append(None)                     # future or no data
            else:
                days.append(is_met(h, by_date[wd]))   # True / False / None
        habits.append({
            "key": h["key"], "name": h["name"], "color": h["color"],
            "type": h.get("type", "number"),
            "days": days, "done": sum(1 for x in days if x),
            "value": getattr(today_log, h["field"], None) if today_log else None,
        })

    # Context inputs (no goal): today's value or None
    context = []
    for c in CONTEXT:
        raw = getattr(today_log, c["field"], None) if today_log else None
        shown = None
        if raw is not None:
            shown = _choice_label(c, raw) or str(raw)
        context.append({"key": c["key"], "name": c["name"],
                        "color": c.get("color", "#6c757d"),
                        "value": raw, "shown": shown})

    elapsed = today.weekday() + 1
    percent = round(sum(h["done"] for h in habits) / (len(HABITS) * elapsed) * 100)

    hour = datetime.now().hour
    greeting = "Good morning" if hour < 12 else "Good afternoon" if hour < 18 else "Good evening"

    return {
        "greeting": greeting,
        "today_label": today.strftime("%a, %b %d"),
        "weeks": weeks,
        "month_labels": labels,
        "habits": habits,
        "context": context,
        "percent": percent,
        "today_index": today.weekday(),
        "today_logged": today_log is not None and level(today_log) > 0,
        "total_logs": sum(1 for l in logs if level(l) > 0),
    }

# Quick-pick values shown as chips on the log page, by habit key. Anything not
# listed falls back to the full grid when it is short (<= 12 values).
PRESETS = {
    "study": [0, 1, 2, 3, 4, 5, 6],
    "sleep": [5, 6, 6.5, 7, 7.5, 8, 9],
    "disruptions": [0, 1, 2, 3, 4, 5],
    "free_hours": [0, 1, 2, 4, 6, 8, 10],
    "stress": list(range(1, 11)),
    "sessions": [0, 1, 2, 3, 4, 5, 6],
}
CHIP_LIMIT = 6            # dropdown habits with more choices keep a <select>


def _goal(habit, value):
    """True / False if `value` meets the habit's goal, None if it has no goal.

    Uses the app's own is_met(), so it works whatever shape HABITS has.
    """
    try:
        return is_met(habit, SimpleNamespace(**{habit["field"]: value}))
    except Exception:
        return None


def _date_arg():
    """?date=YYYY-MM-DD for the log page (today if missing, bad or future)."""
    raw = request.args.get("date")
    try:
        chosen = date.fromisoformat(raw) if raw else date.today()
    except ValueError:
        chosen = date.today()
    return min(chosen, date.today())


def _grid(habit):
    lo, hi = habit.get("min"), habit.get("max")
    if lo is None or hi is None:
        return []
    step = habit.get("step") or 1
    n = min(int(round((hi - lo) / step)) + 1, 400)
    return [round(lo + i * step, 6) for i in range(n)]


def _log_page_context(habit, choice, chosen, existing_value):
    """Extra data the log page needs: presets, goal table, current value."""
    ctx = {
        "chosen_iso": chosen.isoformat(),
        "today_iso": date.today().isoformat(),
        "yesterday_iso": (date.today() - timedelta(days=1)).isoformat(),
        "chosen_is_today": chosen == date.today(),
        "chip_limit": CHIP_LIMIT,
        "presets": [], "goal_points": [], "has_goal": False, "choice_ok": {},
        "current_value": existing_value, "current_label": None,
    }
    if existing_value is not None:
        if choice:
            ctx["current_label"] = _choice_label(habit, existing_value) or str(existing_value)
        else:
            ctx["current_label"] = f"{existing_value:g} {habit.get('unit', '')}".strip()

    if choice:
        ctx["choice_ok"] = {str(v): _goal(habit, v)
                            for v, _ in habit.get("choices", [])}
        ctx["has_goal"] = any(ok is not None for ok in ctx["choice_ok"].values())
        return ctx

    grid = _grid(habit)
    points = [(v, _goal(habit, v)) for v in grid]
    ctx["has_goal"] = any(ok is not None for _, ok in points)
    if ctx["has_goal"]:
        ctx["goal_points"] = [[v, bool(ok)] for v, ok in points if ok is not None]
    wanted = PRESETS.get(habit["key"]) or (grid if len(grid) <= 12 else [])
    lo, hi = habit.get("min"), habit.get("max")
    ctx["presets"] = [{"value": v, "ok": _goal(habit, v)} for v in wanted
                      if (lo is None or v >= lo) and (hi is None or v <= hi)]
    return ctx

@main_bp.route("/habit/<key>", methods=["GET", "POST"])
@login_required
def log_habit(key):
    habit = BY_KEY.get(key)
    if habit is None or habit.get("type") == "check":
        abort(404)

    choice = _is_choice(habit)
    form = ChoiceHabitForm() if choice else NumberHabitForm()
    if choice:
        form.value.choices = [(str(v), label) for v, label in habit["choices"]]

    chosen = _date_arg() if request.method == "GET" else (
        form.log_date.data or date.today())
    existing_value = None
    if request.method == "GET":
        form.log_date.data = chosen
        existing = _get_log(current_user.id, chosen)
        existing_value = getattr(existing, habit["field"], None) if existing else None
        if existing_value is not None:
            form.value.data = str(existing_value) if choice else existing_value
    elif form.log_date.data:
        existing = _get_log(current_user.id, form.log_date.data)
        existing_value = getattr(existing, habit["field"], None) if existing else None

    if form.validate_on_submit():
        lo, hi = habit.get("min"), habit.get("max")
        if form.log_date.data > date.today():
            form.log_date.errors.append("Date can't be in the future.")
        elif not choice and lo is not None and hi is not None \
                and not lo <= form.value.data <= hi:
            form.value.errors.append(f"Enter a value from {lo:g} to {hi:g}.")
        else:
            try:
                stored = habit["cast"](form.value.data)
            except (TypeError, ValueError):
                form.value.errors.append("That value isn't valid.")
                stored = None
            if stored is not None:
                log = _get_log(current_user.id, form.log_date.data, create=True)
                setattr(log, habit["field"], stored)
                db.session.commit()

                if choice:
                    shown = _choice_label(habit, form.value.data) or form.value.data
                else:
                    shown = f"{form.value.data:g} {habit.get('unit', '')}".strip()

                met = is_met(habit, log) if _goal(habit, stored) is not None else None
                if met is None:                           # context inputs: no goal
                    flash(f"{habit['name']} saved: {shown}.", "success")
                else:
                    flash(f"{habit['name']}: "
                          f"{'goal met' if met else 'goal not met'} ({shown}).",
                          "success" if met else "info")
                return redirect(url_for("main.index"))

    ctx = _log_page_context(habit, choice, chosen, existing_value)
    return render_template("habit_form.html", form=form, habit=habit, **ctx)

import pandas as pd
from app.habits import HABITS, goals_met, is_met
from app.models import ProductivityLog


def load_df(user):
    logs = (ProductivityLog.query.filter_by(user_id=user.id)
            .order_by(ProductivityLog.log_date).all())
    rows = []
    for l in logs:
        row = {"date": pd.Timestamp(l.log_date),
               "score": goals_met(l) / len(HABITS) * 100}
        for h in HABITS:
            row[h["field"]] = getattr(l, h["field"])
            row[h["key"] + "_met"] = is_met(h, l)      # True / False / None
        rows.append(row)
    return pd.DataFrame(rows)


def _streaks(met, dates):
    best = cur = 0
    prev = None
    for ok, d in zip(met, dates):
        if not (ok == True):          # False, None or NaN breaks the streak
            cur = 0
        else:
            cur = cur + 1 if (cur and (d - prev).days == 1) else 1
        best = max(best, cur)
        prev = d
    return cur, best


def habit_stats(df):
    out = []
    for h in HABITS:
        met = df[h["key"] + "_met"]
        known = met.dropna()
        cur, best = _streaks(met, df["date"])
        vals = pd.to_numeric(df[h["field"]], errors="coerce").dropna()
        out.append({
            "key": h["key"], "name": h["name"], "color": h["color"],
            "average": round(float(vals.mean()), 2) if len(vals) else None,
            "met_rate": round(float(known.astype(bool).mean()) * 100, 1) if len(known) else 0,
            "current_streak": cur, "best_streak": best,
        })
    return out


def correlations(df):
    fields = [h["field"] for h in HABITS]
    num = df[fields].apply(pd.to_numeric, errors="coerce")
    c = num.corrwith(df["score"]).dropna()
    c = c.reindex(c.abs().sort_values(ascending=False).index)
    return {k: round(float(v), 3) for k, v in c.items()}


def by_weekday(df):
    order = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"]
    g = df.groupby(df["date"].dt.day_name())["score"].mean().reindex(order).dropna()
    return {k: round(float(v), 1) for k, v in g.items()}


def dashboard_stats(user):
    df = load_df(user)
    if df.empty:
        return {"empty": True}
    return {
        "empty": False,
        "summary": {
            "total_logs": len(df),
            "avg_score": round(float(df["score"].mean()), 1),
            "best_day": str(df.loc[df["score"].idxmax(), "date"].date()),
            "best_score": round(float(df["score"].max()), 1),
        },
        "habits": habit_stats(df),
        "trend": {
            "dates": [str(d.date()) for d in df["date"]],
            "score": [round(float(v), 1) for v in df["score"]],
            "rolling": [round(float(v), 1) for v in df["score"].rolling(7, min_periods=1).mean()],
        },
        "correlations": correlations(df),
        "weekday": by_weekday(df),
    }

  (function () {
    var dateEl = document.getElementById("log_date");
    if (dateEl) {
      dateEl.addEventListener("change", function () {
        if (dateEl.value) location.href = "{{ url_for('main.log_habit', key=habit.key) }}?date=" + dateEl.value;

    var input = document.getElementById("value");
    var pts = document.getElementById("goal-points");
    if (!input || input.tagName !== "INPUT") return;

    var step = parseFloat(input.step) || 1;
    var min = input.min !== "" ? parseFloat(input.min) : -Infinity;
    var max = input.max !== "" ? parseFloat(input.max) : Infinity;
    var points = pts ? JSON.parse(pts.textContent || "[]") : [];
    var msg = document.getElementById("goal-msg");
    var chips = document.querySelectorAll(".preset");

    function fmt(n) { return String(Math.round(n * 1e6) / 1e6); }

    function sync() {
      var v = parseFloat(input.value);
      chips.forEach(function (c) {
        var on = !isNaN(v) && Math.abs(parseFloat(c.dataset.value) - v) < 1e-6;
        c.classList.toggle("btn-primary", on);
        c.classList.toggle("btn-outline-secondary", !on);
      });
      msg.textContent = "";
      if (isNaN(v) || !points.length) return;
      var hit = null;
      points.forEach(function (p) { if (Math.abs(p[0] - v) < 1e-6) hit = p[1]; });
      if (hit === null) return;
      msg.className = "small mt-2 " + (hit ? "text-success" : "text-secondary");
      msg.textContent = hit ? "\u2713 Meets your goal" : "Goal not met";
    }

    function nudge(dir) {
      var v = parseFloat(input.value);
      if (isNaN(v)) v = dir > 0 ? min === -Infinity ? 0 : min : 0;
      else v = v + dir * step;
      input.value = fmt(Math.min(max, Math.max(min, v)));
      sync();
    }

    document.getElementById("dec").addEventListener("click", function () { nudge(-1); });
    document.getElementById("inc").addEventListener("click", function () { nudge(1); });
    chips.forEach(function (c) {
      c.addEventListener("click", function () { input.value = c.dataset.value; sync(); });
    });
    input.addEventListener("input", sync);
    sync();
  })();

# Evaluation report (synthetic data)

Trained on synthetic (simulated) data. Metrics show how well the models recover the rules written in dummydataset.py, not how well they would predict a real person.

Hold-out: 4 people never seen in training (1940 rows). The hold-out contains only 4 people, so hold-out metrics are noisy (they change a lot with a different split). Prefer the cross-validated numbers, or generate more people in dummydataset.py.

## Regression (xgboost)
- Hold-out R2 = 0.005, MAE = 9.19 points (score range 0-100)
- Cross-validated numbers (more stable) are in metadata.json under regression.candidates

## Classification (xgboost)
- Hold-out accuracy = 0.605, macro one-vs-rest ROC-AUC = 0.762
- Read confusion_matrix.png: the 'low' level is rare, so expect it to be the hardest class.

## Permutation importance (hold-out)
| feature | regression MAE increase | classification F1 drop |
|---|---|---|
| stress_level | 1.138 | 0.044 |
| study_hours | 0.456 | 0.024 |
| caffeine_mg | 0.307 | 0.001 |
| study_sessions | 0.133 | 0.005 |
| journaled | -0.123 | -0.001 |

Importance describes what the fitted model uses. It is not causal and correlated features share credit.

# Project Context: Personal Productivity Intelligence App

## 1. Project Overview

Build a full-stack web application called **Personal Productivity Intelligence App (PPIA)**.

The application allows users to register, log in securely, record daily productivity-related activities, visualize personal trends, and receive machine-learning-based predictions and insights.

The main purpose is to combine software engineering, database management, data analysis, statistics, and machine learning into one complete portfolio project.

This is not merely a CRUD application or a dashboard displaying static charts. Machine learning and experimentation should be central components of the project.

The initial version should use synthetic data for model development. Once the application is functional, it should collect real daily logs from registered users and use those logs for personalized analysis when sufficient data is available.

### Primary objectives

1. Build a complete authentication system.
2. Implement a relational database with proper relationships.
3. Allow users to record, edit, view, and delete their daily productivity logs.
4. Develop an interactive dashboard with charts and statistics.
5. Experiment with multiple machine-learning algorithms.
6. Compare models using appropriate evaluation metrics.
7. Integrate trained models into the application.
8. Provide personalized, explainable insights based on logged data.
9. Keep the architecture modular, maintainable, and suitable for future expansion.

---

## 2. Target Users

Students, developers, and individuals who want to understand their daily productivity patterns.

Each user must have a private account. Users must only be able to access their own productivity records and personal analytics.

The application should use a clean, modern, responsive interface suitable for desktop and mobile browsers.

### 4.2 User Dashboard

After logging in, the user should see a dashboard containing:

- Average daily study hours
- Average sleep duration
- Average screen time
- Recent productivity history
- Productivity trend chart
- Study hours versus productivity chart
- Recent daily logs
- Quick action to add a new log

Do not display fabricated statistics as if they were real user data.

Allow users to create, view, update, and delete daily records.

Each record should include:

- Date
- Study hours
- Sleep hours
- Screen time in hours
- Exercise duration in minutes
- Caffeine consumption in milligrams
- Stress level from 1 to 10
- Number of breaks
- Number of study sessions
- Optional notes

Validate numerical ranges and reject invalid submissions.

Prevent duplicate daily logs for the same user and date unless the application is deliberately extended to support multiple sessions per day.

Use sensible limits for inputs, such as 0â€“24 hours for sleep and 0â€“24 hours for study or screen time.

Create a history page where users can:

- Browse daily logs
- Filter by date range
- Sort by date or productivity score
- Edit or delete entries
- View individual records
- Export their own records as CSV

### 4.5 Analytics

- Daily and weekly productivity trends
- Monthly summaries
- Correlation heatmaps

- Sleep duration versus productivity
- Stress versus productivity
- Exercise versus productivity
- Summary statistics and outlier detection

### Relationships

One user can have many productivity logs and prediction records.

---

## 6. Synthetic Dataset Generation

Create a script named generate_data.py.

Generate an initial dataset of approximately 10,000 synthetic daily observations and save it to:

datasets/productivity.csv

The dataset is intended for controlled machine-learning experiments, not as evidence about real human behavior.

### Initial features

- study_hours
- sleep_hours
- screen_time
- exercise_minutes
- caffeine_mg
- stress_level
- break_count
- study_sessions
- day_of_week
- previous_day_productivity

### Target variables

- productivity_score: continuous value from 0 to 100
- productive_day: binary label derived from productivity_score using a documented threshold

### Data generation requirements

Do not generate every variable independently from a uniform or random distribution.

Use plausible distributions and introduce relationships between features.

Examples:
- Sleep duration should generally fall within a realistic range.
- Study hours and study sessions should have some relationship.
- Stress, sleep, and productivity should have imperfect relationships.
- Exercise duration should include many low or zero values.
- Screen time should vary among observations.
- Consecutive days may exhibit temporal dependence.
- Include random noise to avoid perfectly predictable outcomes.

Create nonlinear effects, such as diminishing returns from excessive study time and a nonlinear relationship between sleep and productivity.

Include some correlated features, occasional outliers, and a small percentage of missing values in selected input columns.

Document the data-generation assumptions and make the random seed configurable for reproducibility.

Avoid including features that directly reveal the target or leak future information.

For example, if predicting a day's productivity, do not use that day's actual productivity score as an input feature.

The generator may create additional synthetic information needed for controlled experiments, but clearly document any fields not intended for the production application.

### Important limitation

A model trained on synthetic data learns the artificial patterns introduced by the generator. Its metrics do not establish that the model will predict real human productivity accurately.

---

## 7. Machine-Learning Experiments

Organize machine-learning code separately from the Flask routes.

Use reusable functions and scikit-learn Pipelines where appropriate.

### 7.1 Regression

Objective: predict productivity_score from daily activity features.

Compare:
1. Linear Regression
2. Random Forest Regressor
3. Gradient Boosting Regressor
4. XGBoost Regressor

Evaluate with:
- MAE
- RMSE
- RÂ²
- Cross-validation

Use an appropriate train/test split. Keep the test set untouched until final evaluation.

For experiments involving sequential daily observations, consider a chronological split instead of randomly mixing future and past records.

Tune hyperparameters using cross-validation on training data only.

Do not assume XGBoost will always perform best.

### 7.2 Classification

Objective: predict whether a day will meet the defined productivity threshold.

Compare:
1. Logistic Regression
2. Random Forest Classifier
3. Gradient Boosting Classifier
4. XGBoost Classifier

Evaluate with:
- Accuracy
- Precision
- Recall
- F1-score
- ROC-AUC
- Confusion matrix
- Precision-recall curve when class imbalance makes it useful

If productive and unproductive days are imbalanced, report class distribution and use appropriate evaluation methods.

If classification labels are generated from productivity_score, exclude productivity_score from the classification features. Otherwise, the model would receive information that directly determines the label.

### 7.3 Clustering

Objective: discover patterns among daily activity records without using productivity targets.

"""
dummydataset.py: synthetic daily productivity dataset for PPIA.

Run from the project root (as a module, never the VS Code Run button):
    python -m ml.dummydataset
    python -m ml.dummydataset --seed 7 --rows 5000

Writes datasets/productivity_logs.csv (the path set in ml/settings.py).

WHAT THIS IS
    Controlled experiment data, NOT evidence about real human behaviour.
    A model trained on this learns the artificial rules written below.

HOW IT IS BUILT
    * N_PEOPLE synthetic people, each simulated for consecutive calendar days.
    * Each person has a personal baseline and their own logging habit: they
      keep logging with 90% probability, and resume after a miss with 40%.
      Days they did not log are dropped, so log_streak is meaningful.
    * day_type (class / free / deadline) drives free_hours, task_difficulty,
      disruptions and stress. deep_work can never exceed free_hours.
    * Stress follows a slow AR(1) process, so bad days cluster together.
    * productivity_score is a NONLINEAR function of the features plus
      momentum from the previous day and random noise.
    * ~0.5% of rows get extreme outliers; ~3% of values are blanked in four
      input columns to mimic missing data (train.py drops those rows).

MODEL FEATURES (14 inputs, see ml/settings.py)
    Numeric    : deep_work, sleep_hours, caffeine_level, stress_level,
                 journaled, log_streak, disruptions, junk_food,
                 task_difficulty, deep_work_days_prev7, free_hours
    Categorical: day_type, weather        (one-hot encode in the ML code)
    Targets    : productivity_score (0-100), productivity_level (low/medium/high)

FEATURE DEFINITIONS
    log_streak            consecutive logged days ending YESTERDAY, capped at 14.
    deep_work_days_prev7  how many of the previous 7 calendar days met the deep
                          work goal (>= 3h, and were logged). 0-7. Today is
                          excluded. It is a count, not a streak, so one missed
                          day does not reset it. For a person's first week it
                          counts only the days that exist.
    caffeine_level        0-10 scale, mg / 40, so 400 mg = 10 (the app's
                          maximum) and "under 300mg" means a level below 7.5
                          (7 or lower if the app logs whole levels).

LEAKAGE WARNING
    previous_day_productivity is legitimate (known before the day starts) but
    is not a model feature. NEVER use productivity_score, productivity_level
    or productive_day as an input.

EXPERIMENT-ONLY COLUMNS (do not exist in the production app)
    person_id, date, previous_day_productivity.
"""

from pathlib import Path

import pandas as pd

# ----------------------------- defaults ------------------------------------
DEFAULT_SEED = 42

N_PEOPLE = 20

MISSING_RATE = 0.03        # per selected column
OUTLIER_RATE = 0.005       # per row
MISSING_COLUMNS = ["sleep_hours", "caffeine_level", "task_difficulty", "free_hours"]
MG_PER_POINT = 40          # caffeine: 40 mg = 1 point, 400 mg = 10, 300 mg = 7.5

DEEP_WORK_GOAL = 3.0       # hours; same as the app's "Deep work 3h+" habit
STREAK_CAP = 14            # log_streak is capped so it stays a bounded feature
STAY_LOGGING = 0.90        # P(log today | logged yesterday)
RESUME_LOGGING = 0.40      # P(log today | missed yesterday)
DAYS_MARGIN = 1.5          # simulate extra calendar days, missed days are dropped

BASE_SCORE = 43            # overall level of the score formula

DEFAULT_OUTPUT = DATASET_PATH


def simulate_person(rng, n_days, start_date):
    """Return a DataFrame of features for one person (no score yet).

    Every calendar day is simulated; the `logged` column says which days the
    person actually logged. Unlogged days are dropped later in generate().
    """
    dates = pd.date_range(start_date, periods=n_days, freq="D")
    dow = dates.dayofweek.to_numpy()          # 0 = Monday ... 6 = Sunday
    weekend = (dow >= 5).astype(float)

    # Personal baselines
    base_sleep = rng.normal(7.0, 0.5)
    base_deep = rng.normal(3.5, 0.8)
    base_stress = rng.normal(5.0, 0.8)
    base_journal = rng.normal(0.0, 0.12)      # some people journal more

    # Logging behaviour: who logged on which day, and the streak ending yesterday
    logged = np.empty(n_days, dtype=bool)
    logged[0] = True
    for t in range(1, n_days):
        p = STAY_LOGGING if logged[t - 1] else RESUME_LOGGING
        logged[t] = rng.random() < p
    streak_raw = np.zeros(n_days)
    for t in range(1, n_days):
        streak_raw[t] = streak_raw[t - 1] + 1 if logged[t - 1] else 0
    streak = np.minimum(streak_raw, STREAK_CAP)

    # Day type: weekends are mostly free days, weekdays mostly class days
    u = rng.random(n_days)
    day_type = np.where(
        weekend == 1,
        np.where(u < 0.10, "class", np.where(u < 0.80, "free", "deadline")),
        np.where(u < 0.60, "class", np.where(u < 0.75, "free", "deadline")),
    )
    is_class = day_type == "class"
    is_free = day_type == "free"
    is_deadline = (day_type == "deadline").astype(float)

    # Free hours available for work, set mainly by the day type
    free_mean = np.select([is_class, is_free], [5.0, 9.0], default=6.0)
    free = np.clip(free_mean + rng.normal(0, 1.5, n_days), 0, 14)

    # Task difficulty (1-5): harder on deadline days
    difficulty = np.clip(np.round(2.8 + 0.8 * is_deadline + rng.normal(0, 1.0, n_days)), 1, 5)

    # Stress: AR(1) so consecutive days are related, plus deadline and difficulty
    stress_ar = np.empty(n_days)
    stress_ar[0] = base_stress
    for t in range(1, n_days):
        stress_ar[t] = (
            base_stress + 0.8 * (stress_ar[t - 1] - base_stress) + rng.normal(0, 0.9)
        )
    stress = np.clip(np.round(stress_ar + 1.2 * is_deadline + 0.4 * (difficulty - 3)), 1, 10)

    # Disruptions (meetings, interruptions): fewer on weekends
    lam = np.select([is_class, is_free], [1.5, 0.8], default=1.2) * (1 - 0.3 * weekend)
    disruptions = np.clip(rng.poisson(lam), 0, 8).astype(float)

    # Deep work (hours): limited by free hours, cut by disruptions and hard tasks
    deep_work = (
        base_deep * (0.7 + 0.3 * free / 6)
        - 0.35 * disruptions
        - 0.25 * (difficulty - 3)
        + rng.gamma(2.0, 0.6, n_days) - 1.2
    )
    deep_work = np.clip(deep_work, 0, free)
    deep_work[rng.random(n_days) < 0.06] = 0.0     # occasional zero-work days
    deep_work = np.clip(deep_work, 0, 14)

    # Journaling: less likely on stressful days
    p_journal = np.clip(0.55 + base_journal - 0.05 * (stress - 5), 0.05, 0.95)
    journaled = (rng.random(n_days) < p_journal).astype(float)

    # Junk food: more likely when stressed and on weekends
    p_junk = np.clip(0.25 + 0.05 * (stress - 5) + 0.05 * weekend, 0.05, 0.80)
    junk_food = (rng.random(n_days) < p_junk).astype(float)

    # Sleep: weekends longer, stress shortens it
    sleep = (
        base_sleep + 0.6 * weekend - 0.18 * (stress - 5) + rng.normal(0, 0.9, n_days)
    )
    sleep = np.clip(sleep, 3.0, 11.5)

    # Caffeine (mg): rises with deep work and short sleep, some people drink none
    caffeine = 60 + 22 * deep_work + 25 * np.clip(7.5 - sleep, 0, None) + rng.normal(0, 55, n_days)
    caffeine[rng.random(n_days) < 0.12] = 0.0
    caffeine = np.clip(caffeine, 0, 600)

    # Weather: independent of everything else, only a tiny effect on the score
    weather = rng.choice(["sunny", "cloudy", "rainy"], size=n_days, p=[0.5, 0.3, 0.2])

    # Deep-work goal days among the previous 7 calendar days (today excluded).
    # A day counts only if it was logged AND met the goal.
    met = pd.Series(((deep_work >= DEEP_WORK_GOAL) & logged).astype(float))
    prev7 = met.shift(1).rolling(7, min_periods=1).sum().fillna(0).to_numpy()

    return pd.DataFrame(
        {
            "date": dates,
            "day_of_week": dow,
            "logged": logged,
            "deep_work": deep_work,
            "sleep_hours": sleep,
            "caffeine_mg": caffeine,          # renamed to caffeine_level in generate()
            "stress_level": stress,
            "journaled": journaled,
            "log_streak": streak,
            "disruptions": disruptions,
            "junk_food": junk_food,
            "task_difficulty": difficulty,
            "deep_work_days_prev7": prev7,
            "day_type": day_type,
            "weather": weather,
            "free_hours": free,
        }
    )


def add_outliers(df, rng):
    """Replace a few rows' values with extreme but still valid numbers."""
    hit = rng.random(len(df)) < OUTLIER_RATE
    n = int(hit.sum())
    if n == 0:
        return df
    choices = rng.integers(0, 4, size=n)
    idx = df.index[hit]
    for i, c in zip(idx, choices):
        if c == 0:
            df.at[i, "deep_work"] = rng.uniform(12, 18)
        elif c == 1:
            df.at[i, "caffeine_mg"] = rng.uniform(600, 900)
        elif c == 2:
            df.at[i, "sleep_hours"] = rng.uniform(1.5, 3.0)
        else:
            df.at[i, "disruptions"] = rng.integers(10, 16)
    return df


def compute_score(df, rng, person_effect):
    """Sequentially compute the score so each day can depend on the previous one."""
    n = len(df)
    score = np.empty(n)
    prev = np.empty(n)

    deep = df["deep_work"].to_numpy()
    sleep = df["sleep_hours"].to_numpy()
    caffeine = df["caffeine_mg"].to_numpy()        # still in mg here
    stress = df["stress_level"].to_numpy()
    journaled = df["journaled"].to_numpy()
    streak = df["log_streak"].to_numpy()
    disruptions = df["disruptions"].to_numpy()
    junk = df["junk_food"].to_numpy()
    difficulty = df["task_difficulty"].to_numpy()
    prev7 = df["deep_work_days_prev7"].to_numpy()
    day_type = df["day_type"].to_numpy()
    weather = df["weather"].to_numpy()
    free = df["free_hours"].to_numpy()

    # Nonlinear effects (all vectorised)
    sleep_eff = 16 * np.exp(-((sleep - 7.5) / 1.6) ** 2) - 8
    deep_eff = 11 * (1 - np.exp(-deep / 2.5)) - 0.9 * np.clip(deep - 8, 0, None) ** 2
    caffeine_eff = 3 * np.exp(-((caffeine - 150) / 120) ** 2) - 0.015 * np.clip(caffeine - 350, 0, None)
    stress_eff = -2.2 * (stress - 5) - 0.6 * np.clip(stress - 8, 0, None) ** 2
    journal_eff = 3.0 * journaled

    # New features
    use_eff = 4.0 * np.clip(deep / np.maximum(free, 1.0), 0, 1)       # used free time well
    disruption_eff = -1.5 * np.minimum(disruptions, 6)
    junk_eff = -2.0 * junk
    difficulty_eff = -1.0 * (difficulty - 3)
    streak_eff = 0.3 * streak                                          # max +4.2
    prev7_eff = 0.6 * prev7 - 0.5 * np.clip(prev7 - 5, 0, None) ** 2  # fades past 5 (burnout)
    day_type_eff = np.select(
        [day_type == "free", day_type == "deadline"], [-1.5, 1.0], default=0.0
    )
    weather_eff = np.select(
        [weather == "sunny", weather == "rainy"], [0.8, -0.8], default=0.0
    )

    static = (
        BASE_SCORE + person_effect
        + sleep_eff + deep_eff + caffeine_eff + stress_eff + journal_eff
        + use_eff + disruption_eff + junk_eff + difficulty_eff
        + streak_eff + prev7_eff + day_type_eff + weather_eff
    )

    prev_score = 50.0 + person_effect          # day 0 has no real "yesterday"
    for t in range(n):
        prev[t] = prev_score
        s = static[t] + 0.25 * (prev_score - 55) + rng.normal(0, 6)
        score[t] = np.clip(s, 0, 100)
        prev_score = score[t]
    return score, prev


def generate(rows, seed, threshold):
    rng = np.random.default_rng(seed)
    rows_per_person = int(np.ceil(rows / N_PEOPLE))
    days_each = int(np.ceil(rows_per_person * DAYS_MARGIN))
    frames = []

    for pid in range(1, N_PEOPLE + 1):
        person_df = simulate_person(rng, days_each, "2024-01-01")
        person_df = add_outliers(person_df.reset_index(drop=True), rng)

        # Keep values inside the same limits the app's forms will enforce
        person_df["deep_work"] = person_df["deep_work"].clip(0, 24)
        person_df["sleep_hours"] = person_df["sleep_hours"].clip(0, 24)
        person_df["caffeine_mg"] = person_df["caffeine_mg"].clip(0, 1000)

        person_effect = rng.normal(0, 4)
        score, prev = compute_score(person_df, rng, person_effect)
        person_df["productivity_score"] = score
        person_df["previous_day_productivity"] = prev

        # Yesterday is only known if it was a logged day
        person_df.loc[0, "previous_day_productivity"] = np.nan
        yesterday_missing = ~person_df["logged"].shift(1, fill_value=False)
        person_df.loc[yesterday_missing, "previous_day_productivity"] = np.nan

        # Drop the days this person did not log, keep a fixed number of rows
        person_df = person_df[person_df["logged"]].drop(columns="logged")
        person_df = person_df.head(rows_per_person)
        person_df.insert(0, "person_id", pid)
        frames.append(person_df)

    df = pd.concat(frames, ignore_index=True).iloc[:rows].copy()

    # Convert caffeine from mg (used by the score formula) to the app's 0-10 scale
    df["caffeine_level"] = (df["caffeine_mg"] / MG_PER_POINT).clip(0, 10)
    df = df.drop(columns="caffeine_mg")

    # Rounding to realistic precision
    for col in ["deep_work", "sleep_hours", "caffeine_level", "free_hours"]:
        df[col] = df[col].round(1)
    for col in ["stress_level", "journaled", "log_streak", "disruptions",
                "junk_food", "task_difficulty", "deep_work_days_prev7", "day_of_week"]:
        df[col] = df[col].astype(int)
    df["productivity_score"] = df["productivity_score"].round(1)
    df["previous_day_productivity"] = df["previous_day_productivity"].round(1)

    # Labels are derived BEFORE masking so they reflect the true score
    df["productive_day"] = (df["productivity_score"] >= threshold).astype(int)
    df["productivity_level"] = pd.cut(
        df["productivity_score"], bins=LEVEL_BINS, labels=LEVELS
    ).astype(str)

    # Missing values in selected INPUT columns only (never in targets)
    for col in MISSING_COLUMNS:
        mask = rng.random(len(df)) < MISSING_RATE
        df.loc[mask, col] = np.nan

    df["date"] = df["date"].dt.strftime("%Y-%m-%d")

    cols = [
        "person_id", "date", "day_of_week",
        "deep_work", "sleep_hours", "caffeine_level", "stress_level",
        "journaled", "log_streak", "disruptions", "junk_food",
        "task_difficulty", "deep_work_days_prev7", "day_type", "weather",
        "free_hours", "previous_day_productivity",
        "productivity_score", "productivity_level", "productive_day",

"""Evaluate the saved models on the hold-out people and explain them.

Run from the project root, after `python -m ml.train`:
    python -m ml.evaluate

Writes to artifacts/figures/ and artifacts/evaluation_report.md:
    confusion_matrix.png          which levels get confused with which
    regression_pred_vs_actual.png predicted vs real score on unseen people
    permutation_importance.png    what each model relies on
    partial_dependence.png        how the predicted score moves with each input

READ THIS BEFORE QUOTING ANY NUMBER
    * The data is synthetic: results show how well the models recover the rules
      in dummydataset.py, not how well they predict real people.
    * Importance / partial dependence describe the MODEL, not cause and effect.
    * With only a few hold-out people the numbers are noisy; the report says so.
"""
import json

import joblib
import matplotlib

matplotlib.use("Agg")  # draw to files, no window needed
import matplotlib.pyplot as plt
import numpy as np
from sklearn.inspection import PartialDependenceDisplay, permutation_importance
from sklearn.metrics import (ConfusionMatrixDisplay, mean_absolute_error,
                             r2_score, roc_auc_score)

from ml.settings import (ARTIFACT_DIR, CLASSIFICATION_TARGET, FEATURES, LEVELS,
                         RANDOM_STATE, REGRESSION_TARGET)
from ml.train import load_data, split_data

FIG_DIR = ARTIFACT_DIR / "figures"


def main() -> None:
    for name in ("regressor.joblib", "classifier.joblib", "metadata.json"):
        if not (ARTIFACT_DIR / name).exists():
            raise SystemExit("Models not found. Run: python -m ml.train")

    reg = joblib.load(ARTIFACT_DIR / "regressor.joblib")
    clf = joblib.load(ARTIFACT_DIR / "classifier.joblib")
    meta = json.loads((ARTIFACT_DIR / "metadata.json").read_text())

    _, test_df = split_data(load_data())
    if test_df is None:
        raise SystemExit("Dataset too small for a hold-out set (< 100 rows).")
    FIG_DIR.mkdir(parents=True, exist_ok=True)

    X = test_df[FEATURES]
    y_reg = test_df[REGRESSION_TARGET]
    y_clf = test_df[CLASSIFICATION_TARGET].map({n: i for i, n in enumerate(LEVELS)})

    # ---- regression ----
    pred = np.clip(reg.predict(X), 0, 100)
    r2, mae = r2_score(y_reg, pred), mean_absolute_error(y_reg, pred)
    fig, ax = plt.subplots(figsize=(5.5, 5.5))
    ax.scatter(y_reg, pred, s=6, alpha=0.35)
    ax.plot([0, 100], [0, 100], "k--", lw=1)
    ax.set(xlabel="actual score", ylabel="predicted score",
           title=f"Unseen people: R2={r2:.2f}, MAE={mae:.1f}")
    fig.tight_layout()
    fig.savefig(FIG_DIR / "regression_pred_vs_actual.png", dpi=130)
    plt.close(fig)

    # ---- classification ----
    pred_c = clf.predict(X)
    fig, ax = plt.subplots(figsize=(5.5, 4.8))
    ConfusionMatrixDisplay.from_predictions(
        y_clf, pred_c, labels=list(range(len(LEVELS))), display_labels=LEVELS,
        normalize="true", values_format=".0%", ax=ax, colorbar=False)
    ax.set_title("Confusion matrix (share of each true level)")
    fig.tight_layout()
    fig.savefig(FIG_DIR / "confusion_matrix.png", dpi=130)
    plt.close(fig)
    auc = roc_auc_score(y_clf, clf.predict_proba(X), multi_class="ovr",
                        average="macro", labels=list(range(len(LEVELS))))

    # ---- permutation importance (on hold-out, 15 shuffles) ----
    imp_r = permutation_importance(reg, X, y_reg, scoring="neg_mean_absolute_error",
                                   n_repeats=15, random_state=RANDOM_STATE)
    imp_c = permutation_importance(clf, X, y_clf, scoring="f1_macro",
                                   n_repeats=15, random_state=RANDOM_STATE)
    fig, axes = plt.subplots(1, 2, figsize=(11, 4))
    for ax, imp, title in ((axes[0], imp_r, "Regression: MAE increase when shuffled"),
                           (axes[1], imp_c, "Classification: macro-F1 drop when shuffled")):
        order = np.argsort(imp.importances_mean)
        ax.barh(np.array(FEATURES)[order], imp.importances_mean[order],
                xerr=imp.importances_std[order])
        ax.set_title(title)
    fig.tight_layout()
    fig.savefig(FIG_DIR / "permutation_importance.png", dpi=130)
    plt.close(fig)

    # ---- partial dependence of the regressor ----
    fig, ax = plt.subplots(figsize=(13, 3.6))
    # Newer scikit-learn refuses integer columns here, so plot from a float copy.
    PartialDependenceDisplay.from_estimator(reg, X.astype(float), FEATURES, ax=ax, grid_resolution=25)
    fig.suptitle("Partial dependence of predicted score (model behaviour, not causation)")
    fig.tight_layout()
    fig.savefig(FIG_DIR / "partial_dependence.png", dpi=130)
    plt.close(fig)

    # ---- markdown report ----
    split = meta["split"]
    lines = [
        "# Evaluation report (synthetic data)",
        "",
        meta["disclaimer"],
        "",
        f"Hold-out: {split.get('test_people', '?')} people never seen in training "
        f"({split['test_rows']} rows). {split.get('warning', '')}",
        "",
        f"## Regression ({meta['regression']['selected']})",
        f"- Hold-out R2 = {r2:.3f}, MAE = {mae:.2f} points (score range 0-100)",
        "- Cross-validated numbers (more stable) are in metadata.json under "
        "regression.candidates",
        "",
        f"## Classification ({meta['classification']['selected']})",
        f"- Hold-out accuracy = {np.mean(pred_c == y_clf):.3f}, "
        f"macro one-vs-rest ROC-AUC = {auc:.3f}",
        "- Read confusion_matrix.png: the 'low' level is rare, so expect it to be "
        "the hardest class.",
        "",
        "## Permutation importance (hold-out)",
        "| feature | regression MAE increase | classification F1 drop |",
        "|---|---|---|",
    ]
    for i in np.argsort(-imp_r.importances_mean):
        lines.append(f"| {FEATURES[i]} | {imp_r.importances_mean[i]:.3f} | "
                     f"{imp_c.importances_mean[i]:.3f} |")
    lines += ["", "Importance describes what the fitted model uses. It is not causal "
              "and correlated features share credit.", ""]
    (ARTIFACT_DIR / "evaluation_report.md").write_text("\n".join(lines), encoding="utf-8")

    print("\n".join(lines))
    print(f"Figures saved in {FIG_DIR}")


if __name__ == "__main__":

"""Load the saved models and make predictions.

Flask-free on purpose. From the web app:

    from ml.predict import predict_day
    result = predict_day({"deep_work": 4, "sleep_hours": 7, ..., "day_type": "class"})

predict_day needs ALL of settings.FEATURES. Numbers are range-checked; text
features (day_type, weather) must be one of the values seen in training.
The app-side code that turns a database row into these inputs lives in
app/insights/service.py.

Quick test from the project root:
    python -m ml.predict

SECURITY: the .joblib files are pickles. Only load files this project trained.
"""
import json
import math

import joblib
import pandas as pd

import ml.settings as settings
from ml.settings import ARTIFACT_DIR, FEATURES, LEVELS

# Allowed range of every numeric input. settings.FEATURE_RANGES (if present)
# overrides these, but a feature can no longer slip through unchecked just
# because settings.py is out of date.
DEFAULT_RANGES = {
    "deep_work": (0, 24),
    "sleep_hours": (0, 24),
    "caffeine_level": (0, 10),
    "stress_level": (1, 10),
    "journaled": (0, 1),
    "log_streak": (0, 14),
    "disruptions": (0, 20),
    "junk_food": (0, 1),
    "task_difficulty": (1, 5),
    "deep_work_days_prev7": (0, 7),
    "free_hours": (0, 24),
}
FEATURE_RANGES = {**DEFAULT_RANGES, **getattr(settings, "FEATURE_RANGES", {})}

DISCLAIMER = (
    "Experimental: trained on synthetic (simulated) data. This is an estimate "
    "for the day described by these inputs, not a forecast for tomorrow and "
    "not a statement about you."
)

_cache = {}


class ModelsNotTrainedError(RuntimeError):
    pass


def _load():
    """Load models once, then reuse them for every request."""
    if _cache:
        return _cache
    try:
        _cache["regressor"] = joblib.load(ARTIFACT_DIR / "regressor.joblib")
        _cache["classifier"] = joblib.load(ARTIFACT_DIR / "classifier.joblib")
        _cache["kmeans"] = joblib.load(ARTIFACT_DIR / "kmeans.joblib")
        _cache["profiles"] = pd.read_csv(
            ARTIFACT_DIR / "cluster_profiles.csv", index_col="cluster")
        with open(ARTIFACT_DIR / "metadata.json") as f:
            _cache["metadata"] = json.load(f)
    except FileNotFoundError as exc:
        _cache.clear()
        raise ModelsNotTrainedError(
            "Models not found. Run: python -m ml.train") from exc
    return _cache


def reload_models() -> None:
    """Forget cached models (call after retraining without restarting Flask)."""
    _cache.clear()


def _validate(values: dict, meta: dict) -> pd.DataFrame:
    """Check one day's inputs and return a one-row DataFrame in FEATURES order."""
    missing = [f for f in FEATURES if f not in values]
    if missing:
        raise ValueError(f"Missing inputs: {missing}")

    # The models were trained on these exact columns. If settings.FEATURES
    # changed after training, stop instead of predicting from the wrong inputs.
    trained = meta.get("features")
    if trained is not None and list(trained) != list(FEATURES):
        raise ValueError(
            "settings.FEATURES differs from the trained models. "
            "Retrain: python -m ml.train")

    categories = meta.get("categories", {})
    cat = set(meta.get("categorical_features", []))
    row = {}
    for name in FEATURES:
        raw = values[name]
        if name in cat:
            text = "" if raw is None else str(raw)
            allowed = categories.get(name)
            if allowed and text not in allowed:
                raise ValueError(
                    f"'{name}' must be one of: {', '.join(allowed)}")
            row[name] = text
            continue
        try:
            value = float(raw)
        except (TypeError, ValueError):
            raise ValueError(f"'{name}' must be a number")
        if math.isnan(value):
            raise ValueError(f"'{name}' must be a number")
        bounds = FEATURE_RANGES.get(name)
        if bounds is not None:
            low, high = bounds
            if not low <= value <= high:
                raise ValueError(f"'{name}' must be between {low} and {high}")
        row[name] = value
    return pd.DataFrame([row], columns=FEATURES)  # fixed column order


def predict_day(values: dict) -> dict:
    """Predict score, level and behaviour cluster for one day."""
    models = _load()
    meta = models["metadata"]
    X = _validate(values, meta)

    score = float(models["regressor"].predict(X)[0])
    score = round(min(max(score, 0.0), 100.0), 1)

    clf = models["classifier"]
    class_id = int(clf.predict(X)[0])
    probs = clf.predict_proba(X)[0]
    cluster = int(models["kmeans"].predict(X)[0])

    reg_meta = meta.get("regression") or {}
    # Typical error = cross-validated MAE of the selected model (steadier than
    # a hold-out made of only a few people).
    cv_mae = next((c["cv_mae"] for c in reg_meta.get("candidates", [])
                   if c["model"] == reg_meta.get("selected")), None)
    return {
        "predicted_score": score,
        "predicted_level": LEVELS[class_id],
        "level_probabilities": {
            LEVELS[int(c)]: round(float(p), 3)
            for c, p in zip(clf.classes_, probs)
        },
        "cluster": cluster,
        "cluster_profile": models["profiles"].loc[cluster].to_dict(),
        "typical_error_points": cv_mae,
        "model_version": meta.get("model_version"),
        "experimental": True,
        "disclaimer": DISCLAIMER,
    }


# What-if levers: (feature, value to try, wording). suggest_levers() re-runs the
# regressor with one input changed and reports how much the estimate moves.
# This is what the MODEL does, not proof that the change causes anything.
LEVERS = [
    ("sleep_hours", 7.5, "Sleep about 7.5 h"),
    ("deep_work", 4, "Reach about 4 h of deep work"),
    ("stress_level", 3, "Bring stress down to 3"),
    ("disruptions", 0, "Have no disruptions"),
    ("junk_food", 0, "Skip junk food"),
    ("journaled", 1, "Journal today"),
    ("caffeine_level", 3, "Keep caffeine around level 3"),
]


def suggest_levers(values: dict, top: int = 3, min_gain: float = 0.5) -> list:
    """Biggest single changes (by the model) that would raise the score."""
    models = _load()
    meta = models["metadata"]
    base = _validate(values, meta)

    rows, info = [], []
    for feature, target, text in LEVERS:
        if feature not in FEATURES:
            continue
        try:
            if float(values[feature]) == target:
                continue
        except (TypeError, ValueError, KeyError):
            continue
        rows.append(_validate({**values, feature: target}, meta))
        info.append((feature, text))
    if not rows:
        return []

    preds = models["regressor"].predict(pd.concat([base] + rows, ignore_index=True))
    found = []
    for (feature, text), p in zip(info, preds[1:]):
        gain = float(p - preds[0])
        if gain >= min_gain:
            found.append({"feature": feature, "text": text, "gain": round(gain, 1)})
    return sorted(found, key=lambda d: -d["gain"])[:top]


def predict_for_log(log) -> dict:
    """Predict from any object whose attributes are named like FEATURES.

    This does NOT work on a raw ProductivityLog (its column names differ and
    two features are computed from past days). For the app use
    app.insights.service.latest_estimate instead.
    """
    values = {name: getattr(log, name, None) for name in FEATURES}
    missing = [name for name, v in values.items() if v is None]
    if missing:
        raise ValueError("This log is missing: " + ", ".join(missing))
    values = {name: int(v) if isinstance(v, bool) else v
              for name, v in values.items()}
    return predict_day(values)


def data_stage(n_logs: int) -> str:
    """How much of the user's own data exists decides what the app may claim.

    No stage means "a model trained on you": there is no per-user model.
    """
    if n_logs < 7:
        return "general"            # only the synthetic-trained estimate
    if n_logs < 20:
        return "descriptive"        # also show the user's own averages
    return "enough_for_trends"      # enough rows to show trends from their logs


def model_info() -> dict:
    """Metrics, split and feature importance, for an ML Insights page."""
    return _load()["metadata"]

===== ml/train.py =====

"""Train, compare and save the three models.

Run from the project root:
    python -m ml.train

Saves to artifacts/:
    regressor.joblib, classifier.joblib, kmeans.joblib,
    metadata.json, cluster_profiles.csv

FEATURE TYPES
    FEATURES (in ml/settings.py) may mix numbers and text. Any feature whose
    column is not numeric in the data (day_type, weather) is one-hot encoded
    inside each model's pipeline; numbers are scaled for the linear models and
    clustering. Because the encoding is part of the saved pipeline, a
    prediction needs a one-row DataFrame with the FEATURES column names
    (not a plain list or array).

VALIDATION (what changed vs the first version)
    Rows from the same person are strongly related (the score carries momentum
    from the previous day). A shuffled split would put the same person in both
    train and test and flatter the metrics. So when the dataset has a
    person_id column:
        * the hold-out set is made of WHOLE PEOPLE the model never saw
          (GroupShuffleSplit), which also matches the app's cold-start case:
          a brand-new user the model has never met
        * cross-validation uses GroupKFold (same idea inside the training part)
    Without person_id (e.g. real logs later) it falls back to the old
    shuffled/stratified behaviour.

    rows < SMALL_N       : no hold-out, repeated k-fold CV only, no XGBoost
    rows >= SMALL_N      : 80/20 hold-out + 5-fold CV on the training part
    rows >= XGB_MIN_ROWS : XGBoost joins the candidates

The winner is chosen on CROSS-VALIDATION scores only. Hold-out numbers are
reported, never used to choose.
"""

import platform
from datetime import datetime, timezone

import joblib
import numpy as np
import pandas as pd
import sklearn
from pandas.api.types import is_numeric_dtype
from sklearn.cluster import KMeans
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import RandomForestClassifier, RandomForestRegressor
from sklearn.linear_model import LogisticRegression, Ridge
from sklearn.metrics import (accuracy_score, classification_report,
                             confusion_matrix, f1_score, mean_absolute_error,
                             mean_squared_error, r2_score, silhouette_score)

# Columns that must never be model inputs (they are, or come from, the answer).
LEAKY = {"productivity_score", "productivity_level", "productive_day",
         "previous_day_productivity"}


# ------------------------------------------------------ feature preprocessing
def split_feature_types(df: pd.DataFrame):
    """Return (numeric, categorical) lists, in FEATURES order.

    A feature is categorical when its column is not numeric in the data
    (day_type, weather). Those are one-hot encoded; the rest are numbers.
    """
    cat = [c for c in FEATURES if not is_numeric_dtype(df[c])]
    num = [c for c in FEATURES if c not in cat]
    return num, cat


def make_prep(num, cat, scale: bool) -> ColumnTransformer:
    """Numbers are scaled (linear models / clustering) or passed through
    (trees); categoricals are one-hot encoded. Unknown categories at
    prediction time become all zeros instead of crashing."""
    parts = [("num", StandardScaler() if scale else "passthrough", num)]
    if cat:
        parts.append(("cat", OneHotEncoder(handle_unknown="ignore",
                                           sparse_output=False), cat))
    return ColumnTransformer(parts)


def _original_feature(name: str, cat) -> str:
    """'cat__day_type_class' -> 'day_type', 'num__deep_work' -> 'deep_work'."""
    name = name.split("__", 1)[-1]
    for c in cat:
        if name.startswith(c + "_"):
            return c
    return name


def feature_importance(model, cat):
    """Tree importances summed back to the original features (one-hot
    columns of day_type are added up into a single 'day_type' value)."""
    est = model.named_steps["model"]
    if not hasattr(est, "feature_importances_"):
        return None
    names = model.named_steps["prep"].get_feature_names_out()
    out = {}
    for n, v in zip(names, est.feature_importances_):
        k = _original_feature(n, cat)
        out[k] = out.get(k, 0.0) + float(v)
    return {k: round(v, 4) for k, v in sorted(out.items(), key=lambda kv: -kv[1])}


def linear_coefficients(model):
    """Standardised coefficients of a linear regressor (sign = direction)."""
    est = model.named_steps["model"]
    if not hasattr(est, "coef_") or np.ndim(est.coef_) != 1:
        return None
    names = model.named_steps["prep"].get_feature_names_out()
    pairs = {n.split("__", 1)[-1]: round(float(c), 4)
             for n, c in zip(names, est.coef_)}
    return dict(sorted(pairs.items(), key=lambda kv: -abs(kv[1])))


# --------------------------------------------------------------------- data
def load_data() -> pd.DataFrame:
    if not DATASET_PATH.exists():
        raise SystemExit(
            f"Dataset not found at {DATASET_PATH}.\n"
            "Generate it first:  python -m ml.dummydataset --rows 10000"
        )
    leaky = sorted(LEAKY & set(FEATURES))
    if leaky:
        raise SystemExit(f"FEATURES must not contain answer columns: {leaky}")
    df = pd.read_csv(DATASET_PATH)
    needed = FEATURES + [REGRESSION_TARGET, CLASSIFICATION_TARGET]
    missing = [c for c in needed if c not in df.columns]
    if missing:
        raise SystemExit(f"Dataset is missing columns: {missing}")
    return df.dropna(subset=needed).reset_index(drop=True)


def split_data(df: pd.DataFrame):
    """Return (train_df, test_df). test_df is None in small-data mode.

    Deterministic (fixed RANDOM_STATE) so evaluate.py rebuilds the same split.
    """
    if len(df) < SMALL_N:
        return df, None
    if GROUP_COL in df.columns:
        splitter = GroupShuffleSplit(n_splits=1, test_size=TEST_SIZE,
                                     random_state=RANDOM_STATE)
        train_idx, test_idx = next(splitter.split(df, groups=df[GROUP_COL]))
        return df.iloc[train_idx].reset_index(drop=True), \
            df.iloc[test_idx].reset_index(drop=True)
    train_df, test_df = train_test_split(
        df, test_size=TEST_SIZE, random_state=RANDOM_STATE,
        stratify=df[CLASSIFICATION_TARGET])
    return train_df.reset_index(drop=True), test_df.reset_index(drop=True)


def split_description(train_df, test_df) -> dict:
    grouped = GROUP_COL in train_df.columns
    desc = {
        "method": ("hold-out of whole people (GroupShuffleSplit) + GroupKFold CV"
                   if grouped and test_df is not None else
                   "no hold-out; repeated k-fold CV" if test_df is None else
                   "stratified shuffled hold-out + KFold CV"),
        "train_rows": int(len(train_df)),
        "test_rows": 0 if test_df is None else int(len(test_df)),
    }
    if grouped:
        desc["train_people"] = int(train_df[GROUP_COL].nunique())
        desc["test_people"] = (0 if test_df is None
                               else int(test_df[GROUP_COL].nunique()))
        if test_df is not None and desc["test_people"] < 10:
            desc["warning"] = (
                f"The hold-out contains only {desc['test_people']} people, so "
                "hold-out metrics are noisy (they change a lot with a different "
                "split). Prefer the cross-validated numbers, or generate more "
                "people in dummydataset.py.")
    return desc


def use_xgb(n_rows: int) -> bool:
    return HAS_XGB and n_rows >= XGB_MIN_ROWS


def _groups(train_df):
    return train_df[GROUP_COL] if GROUP_COL in train_df.columns else None


# ---------------------------------------------------------------- regression
def regression_candidates(n_rows: int, num, cat) -> dict:
    small = n_rows < SMALL_N
    candidates = {
        "ridge": Pipeline([("prep", make_prep(num, cat, scale=True)),
                           ("model", Ridge(alpha=1.0))]),
        "random_forest": Pipeline([
            ("prep", make_prep(num, cat, scale=False)),
            ("model", RandomForestRegressor(
                n_estimators=300,
                max_depth=4 if small else None,
                min_samples_leaf=5 if small else 3,
                random_state=RANDOM_STATE, n_jobs=-1))]),
    }
    if use_xgb(n_rows):
        candidates["xgboost"] = Pipeline([
            ("prep", make_prep(num, cat, scale=False)),
            ("model", XGBRegressor(
                n_estimators=400, learning_rate=0.05, max_depth=4,
                subsample=0.9, random_state=RANDOM_STATE, n_jobs=-1))])
    return candidates


def regression_metrics(y_true, pred) -> dict:
    return {"r2": round(float(r2_score(y_true, pred)), 4),
            "mae": round(float(mean_absolute_error(y_true, pred)), 3),
            "rmse": round(float(np.sqrt(mean_squared_error(y_true, pred))), 3)}


def train_regression(X_train, y_train, X_test=None, y_test=None, groups=None,
                     num=(), cat=()):
    n = len(X_train) + (0 if X_test is None else len(X_test))
    if groups is not None:
        cv = GroupKFold(n_splits=min(5, groups.nunique()))
    elif n < SMALL_N:
        cv = RepeatedKFold(n_splits=5, n_repeats=3, random_state=RANDOM_STATE)
    else:
        cv = KFold(5, shuffle=True, random_state=RANDOM_STATE)
    scoring = {"r2": "r2", "mae": "neg_mean_absolute_error"}

    rows, fitted = [], {}
    for name, model in regression_candidates(n, num, cat).items():
        res = cross_validate(model, X_train, y_train, cv=cv,
                             groups=groups, scoring=scoring)
        model.fit(X_train, y_train)
        row = {"model": name,
               "cv_r2": round(float(res["test_r2"].mean()), 4),
               "cv_mae": round(float(-res["test_mae"].mean()), 3),
               "test_r2": None, "test_mae": None}
        if X_test is not None:
            m = regression_metrics(y_test, model.predict(X_test))
            row["test_r2"], row["test_mae"] = m["r2"], m["mae"]
        rows.append(row)
        fitted[name] = model
    results = pd.DataFrame(rows).sort_values("cv_r2", ascending=False)
    best = results.iloc[0]["model"]
    return fitted[best], best, results


# ------------------------------------------------------------ classification
def classification_candidates(n_rows: int, num, cat) -> dict:
    small = n_rows < SMALL_N
    candidates = {
        "logistic_regression": Pipeline([
            ("prep", make_prep(num, cat, scale=True)),
            ("model", LogisticRegression(max_iter=1000))]),
        "random_forest": Pipeline([
            ("prep", make_prep(num, cat, scale=False)),
            ("model", RandomForestClassifier(
                n_estimators=300,
                max_depth=4 if small else None,
                min_samples_leaf=5 if small else 3,
                random_state=RANDOM_STATE, n_jobs=-1))]),
    }
    if use_xgb(n_rows):
        candidates["xgboost"] = Pipeline([
            ("prep", make_prep(num, cat, scale=False)),
            ("model", XGBClassifier(
                n_estimators=300, learning_rate=0.05, max_depth=4,
                subsample=0.9, random_state=RANDOM_STATE,
                eval_metric="mlogloss", n_jobs=-1))])
    return candidates


def train_classification(X_train, y_train, X_test=None, y_test=None,
                         groups=None, num=(), cat=()):
    n = len(X_train) + (0 if X_test is None else len(X_test))
    min_class = int(pd.Series(y_train).value_counts().min())
    if min_class < 2:
        raise SystemExit(
            "A level has fewer than 2 rows; cannot cross-validate the "
            "classifier. Log more days or merge levels.")
    if groups is not None:
        cv = GroupKFold(n_splits=min(5, groups.nunique()))
    else:
        n_splits = min(5, min_class)
        cv = (RepeatedStratifiedKFold(n_splits=n_splits, n_repeats=3,
                                      random_state=RANDOM_STATE)
              if n < SMALL_N else StratifiedKFold(
                  n_splits, shuffle=True, random_state=RANDOM_STATE))
    scoring = {"f1": "f1_macro", "acc": "accuracy"}

    rows, fitted = [], {}
    for name, model in classification_candidates(n, num, cat).items():
        res = cross_validate(model, X_train, y_train, cv=cv,
                             groups=groups, scoring=scoring)
        model.fit(X_train, y_train)
        row = {"model": name,
               "cv_f1_macro": round(float(res["test_f1"].mean()), 4),
               "cv_accuracy": round(float(res["test_acc"].mean()), 4),
               "test_accuracy": None, "test_f1_macro": None}
        if X_test is not None:
            pred = model.predict(X_test)
            row["test_accuracy"] = round(float(accuracy_score(y_test, pred)), 4)
            row["test_f1_macro"] = round(
                float(f1_score(y_test, pred, average="macro")), 4)
        rows.append(row)
        fitted[name] = model
    results = pd.DataFrame(rows).sort_values("cv_f1_macro", ascending=False)
    best = results.iloc[0]["model"]
    return fitted[best], best, results


# ------------------------------------------------------------------ clusters
def train_clustering(df: pd.DataFrame, num, cat):
    """Scale numbers + one-hot categoricals, pick k by silhouette, fit K-Means.

    Unsupervised and uses FEATURES only (no productivity columns), so it is
    fitted on all rows. Silhouette is a guide: check the profiles make sense.
    """
    prep = make_prep(num, cat, scale=True)
    X_t = prep.fit_transform(df[FEATURES])

    ks = [k for k in N_CLUSTERS_RANGE if k < len(df)]
    scores = {}
    for k in ks:
        labels = KMeans(n_clusters=k, n_init=10,
                        random_state=RANDOM_STATE).fit_predict(X_t)
        scores[k] = silhouette_score(
            X_t, labels, sample_size=min(4000, len(df)),
            random_state=RANDOM_STATE)
    best_k = max(scores, key=scores.get)

    kmeans = KMeans(n_clusters=best_k, n_init=10, random_state=RANDOM_STATE)
    labels = kmeans.fit_predict(X_t)

    # Profiles: mean of each numeric feature, most common value of each
    # categorical one, and the average score (for reading, not for fitting).
    profiles = (df[num + [REGRESSION_TARGET]].assign(cluster=labels)
                .groupby("cluster").mean().round(2))
    if cat:
        modes = (df[cat].assign(cluster=labels).groupby("cluster")
                 .agg(lambda s: s.mode().iloc[0]))
        profiles = profiles.join(modes)
    profiles.insert(0, "days", pd.Series(labels).value_counts().sort_index())
    pipeline = Pipeline([("prep", prep), ("kmeans", kmeans)])
    return pipeline, best_k, scores, profiles


# ---------------------------------------------------------------------- main
def main() -> None:
    df = load_data()
    n = len(df)
    num, cat = split_feature_types(df)
    print(f"Loaded {n} rows, {len(FEATURES)} features "
          f"({len(num)} numeric, {len(cat)} categorical: {cat})")
    if n < SMALL_N:
        print(f"Small-data mode (< {SMALL_N} rows): repeated CV, "
              "no hold-out, no XGBoost.\n")
    else:
        print(f"XGBoost {'ON' if use_xgb(n) else 'OFF'} "
              f"(needs >= {XGB_MIN_ROWS} rows and the package installed)\n")

    level_to_id = {name: i for i, name in enumerate(LEVELS)}
    train_df, test_df = split_data(df)
    split_info = split_description(train_df, test_df)
    print("Split:", split_info, "\n")

    X_train = train_df[FEATURES]
    yr_train = train_df[REGRESSION_TARGET]
    yc_train = train_df[CLASSIFICATION_TARGET].map(level_to_id)
    groups = _groups(train_df)
    if test_df is None:
        X_test = yr_test = yc_test = None
    else:
        X_test = test_df[FEATURES]
        yr_test = test_df[REGRESSION_TARGET]
        yc_test = test_df[CLASSIFICATION_TARGET].map(level_to_id)

    print("=== Regression: predict productivity_score ===")
    reg, reg_name, reg_results = train_regression(
        X_train, yr_train, X_test, yr_test, groups, num, cat)
    print(reg_results.to_string(index=False))
    print(f"Best (by CV R2): {reg_name}\n")

    print("=== Classification: predict productivity_level ===")
    clf, clf_name, clf_results = train_classification(
        X_train, yc_train, X_test, yc_test, groups, num, cat)
    print(clf_results.to_string(index=False))
    print(f"Best (by CV macro-F1): {clf_name}\n")

    print("=== Clustering: K-Means ===")
    km, best_k, sil_scores, profiles = train_clustering(df, num, cat)
    for k, s in sil_scores.items():
        print(f"k={k}: silhouette={s:.3f}")
    print(f"Chosen k = {best_k}\n")
    print(profiles.to_string(), "\n")

    # Hold-out metrics for the selected models (reporting only).
    reg_test = clf_test = None
    if X_test is not None:
        reg_test = regression_metrics(yr_test, reg.predict(X_test))
        pred = clf.predict(X_test)
        ids = list(range(len(LEVELS)))
        clf_test = {
            "accuracy": round(float(accuracy_score(yc_test, pred)), 4),
            "f1_macro": round(float(f1_score(yc_test, pred, average="macro")), 4),
            "confusion_matrix": confusion_matrix(
                yc_test, pred, labels=ids).tolist(),
            "labels": LEVELS,
            "per_class": classification_report(
                yc_test, pred, labels=ids, target_names=LEVELS,
                output_dict=True, zero_division=0),
        }
        print("Hold-out (selected models):", reg_test,
              {k: clf_test[k] for k in ("accuracy", "f1_macro")}, "\n")

    ARTIFACT_DIR.mkdir(parents=True, exist_ok=True)
    joblib.dump(reg, ARTIFACT_DIR / "regressor.joblib")
    joblib.dump(clf, ARTIFACT_DIR / "classifier.joblib")
    joblib.dump(km, ARTIFACT_DIR / "kmeans.joblib")
    profiles.to_csv(ARTIFACT_DIR / "cluster_profiles.csv")

    importance = {}
    imp = feature_importance(reg, cat)
    if imp:
        importance["regressor"] = imp
    imp = feature_importance(clf, cat)
    if imp:
        importance["classifier"] = imp
    coefficients = linear_coefficients(reg)

    now = datetime.now(timezone.utc)
    metadata = {
        "model_version": now.strftime("%Y%m%d-%H%M%S"),
        "trained_at_utc": now.isoformat(),
        "sklearn_version": sklearn.__version__,
        "python_version": platform.python_version(),
        "experimental": True,
        "disclaimer": DISCLAIMER,
        "features": FEATURES,
        "numeric_features": num,
        "categorical_features": cat,
        "categories": {c: sorted(df[c].astype(str).unique().tolist()) for c in cat},
        "levels": LEVELS,
        "rows_trained_on": int(n),
        "split": split_info,
        "regression": {
            "selected": reg_name,
            "selection_rule": "highest cross-validated R2 on the training part",
            "candidates": reg_results.to_dict("records"),
            "holdout": reg_test,
        },
        "classification": {
            "selected": clf_name,
            "selection_rule": "highest cross-validated macro-F1 on the training part",
            "candidates": clf_results.to_dict("records"),
            "holdout": clf_test,
        },
        "kmeans": {"k": int(best_k),
                   "silhouette": round(float(sil_scores[best_k]), 4),
                   "silhouette_by_k": {str(k): round(float(s), 4)
                                       for k, s in sil_scores.items()}},
        "feature_importance": importance,
        "regressor_coefficients": coefficients,
        "random_state": RANDOM_STATE,
    }
    with open(ARTIFACT_DIR / "metadata.json", "w") as f:
        json.dump(metadata, f, indent=2, default=str)
    print(f"Saved models and metadata to {ARTIFACT_DIR}")


if __name__ == "__main__":
    main()
